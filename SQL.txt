CREATE DATABASE AUTOSERVICE;
GO

USE AUTOSERVICE;
GO
 
-- =======================================================
-- 3. Tabla de Usuarios (Para el Login)
-- =======================================================
CREATE TABLE Usuario (
    ID INT IDENTITY(1,1) PRIMARY KEY,
    USUARIO VARCHAR(50) NOT NULL UNIQUE,
    PASSWORD VARCHAR(255) NOT NULL, -- Almacenará la contraseña (idealmente encriptada)
    FECHA_REGISTRO DATETIME DEFAULT GETDATE()
);
GO



USE AUTOSERVICE;
GO

CREATE TABLE Cliente (
    ID INT IDENTITY(1,1) PRIMARY KEY,
    NOMBRE VARCHAR(50) NOT NULL,
    APELLIDO VARCHAR(50) NOT NULL,
    DNI VARCHAR(15) NOT NULL UNIQUE,
    TELEFONO VARCHAR(30) NULL,
    EMAIL VARCHAR(100) NULL,
    FECHA_REGISTRO DATETIME DEFAULT GETDATE()
);
GO

CREATE TABLE Vehiculo (
    ID INT IDENTITY(1,1) PRIMARY KEY,
    ID_CLIENTE INT NOT NULL FOREIGN KEY REFERENCES Cliente(ID),
    PATENTE VARCHAR(10) NOT NULL UNIQUE,
    MARCA VARCHAR(50) NOT NULL,
    MODELO VARCHAR(50) NOT NULL,
    ANIO INT NOT NULL CHECK (ANIO BETWEEN 1950 AND 2100),
    COLOR VARCHAR(30) NULL,
	KILOMETRAJE INT NOT NULL DEFAULT 0,
    FECHA_REGISTRO DATETIME DEFAULT GETDATE()
);
GO

CREATE INDEX IX_Vehiculo_Cliente ON Vehiculo(ID_CLIENTE);
GO


CREATE TABLE Repuesto (
    ID INT IDENTITY(1,1) PRIMARY KEY,
    CODIGO VARCHAR(50) NOT NULL UNIQUE,
    DESCRIPCION VARCHAR(150) NOT NULL,
    PRECIO DECIMAL(18,2) NOT NULL DEFAULT 0 CHECK (PRECIO >= 0),
    STOCK_FISICO INT NOT NULL DEFAULT 0 CHECK (STOCK_FISICO >= 0)
);
GO

CREATE INDEX IX_Repuesto_Descripcion ON Repuesto(DESCRIPCION);
GO
 

CREATE TABLE OrdenServicio (
    ID INT IDENTITY(1,1) PRIMARY KEY,
    ID_VEHICULO INT NOT NULL FOREIGN KEY REFERENCES Vehiculo(ID),
    FECHA_INGRESO DATETIME NOT NULL DEFAULT GETDATE(),
	ID_TECNICO INT NOT NULL,
    FECHA_CIERRE DATETIME NULL,
    DIAGNOSTICO VARCHAR(500) NULL,
    ESTADO VARCHAR(50) NOT NULL,
    MANO_OBRA DECIMAL(18,2) NOT NULL DEFAULT 0, 
    TOTAL DECIMAL(18,2) NOT NULL DEFAULT 0,
	TOTAL_REPUESTOS DECIMAL(18,2) NOT NULL DEFAULT 0,
	KILOMETRAJE INT NOT NULL DEFAULT 0
);
GO

CREATE TABLE DetalleOrden (
    ID INT IDENTITY(1,1) PRIMARY KEY,
    ID_ORDEN INT NOT NULL FOREIGN KEY REFERENCES OrdenServicio(ID),
    ID_REPUESTO INT NOT NULL FOREIGN KEY REFERENCES Repuesto(ID),
    CANTIDAD INT NOT NULL CHECK (CANTIDAD > 0),
    PRECIO_UNITARIO DECIMAL(18,2) NOT NULL CHECK (PRECIO_UNITARIO >= 0)
);
GO
